// 6a. Ruta del dinero (solo transferencias, respetando la dirección)
MATCH (a:Cuenta {cuenta_id: 'CTA00001'}), (b:Cuenta {cuenta_id: 'CTA01544'}) // Ubica los nodos de inicio (CTA00001) y destino (CTA01544).
MATCH p = shortestPath((a)-[:TRANSFIERE_A*..6]->(b)) // Busca el camino más corto entre ambas cuentas siguiendo únicamente transferencias salientes dirigidas con un //máximo de hasta 6 saltos
RETURN [n IN nodes(p) | n.cuenta_id] AS ruta, // Itera sobre los nodos del camino p extrayendo una lista secuencial de IDs de cuentas.
       [r IN relationships(p) | r.monto] AS montos, // Extrae la lista de montos y fechas de cada transferencia que compone la ruta.
       [r IN relationships(p) | r.fecha] AS fechas, 
       length(p) AS saltos; // Cuenta cuántas transferencias intermedias se necesitaron para conectar ambas cuentas.

// 6b. Conexión por cualquier vínculo (transferencias, dispositivos, IPs o mismo dueño)
MATCH (a:Cuenta {cuenta_id: 'CTA00001'}), (b:Cuenta {cuenta_id: 'CTA01544'}) // Ubica los nodos de inicio (CTA00001) y destino (CTA01544).
MATCH p = shortestPath((a)-[:TRANSFIERE_A|USA_DISPOSITIVO|CONECTA_DESDE|POSEE*..8]-(b)) // Busca el camino más corto sin importar la dirección y admitiendo cualquier // tipo de vínculo (dinero, dispositivo compartido, dirección IP compartida o titular común/POSEE) hasta un límite de 8 saltos.
RETURN [n IN nodes(p) | labels(n)[0] + ' ' + coalesce(n.cuenta_id, n.dispositivo_id, n.ip_address, n.cliente_id)] AS ruta, //Como los nodos intermedios pueden ser de // distinta etiqueta (Cuenta, Dispositivo, IP, Cliente), coalesce toma el identificador correspondiente que no sea nulo y lo concatena con la etiqueta (labels(n)[0]).
       [r IN relationships(p) | type(r)] AS vinculos, // Devuelve los nombres de las relaciones que unieron a las entidades a lo largo de la cadena.
       length(p) AS saltos;

// 6c. Visualización del camino
MATCH (a:Cuenta {cuenta_id: 'CTA00001'}), (b:Cuenta {cuenta_id: 'CTA01544'})
MATCH p = shortestPath((a)-[:TRANSFIERE_A|USA_DISPOSITIVO|CONECTA_DESDE|POSEE*..8]-(b))
RETURN p; // Hace exactamente la misma búsqueda que la consulta 6b, pero al hacer RETURN p, Neo4j Browser renderiza visualmente el grafo completo con todos los nodos y // aristas interactivos en pantalla.

// 7. RANKING de riesgo (sale como tabla)
MATCH (c:Cuenta)
OPTIONAL MATCH (:Cuenta)-[t:TRANSFIERE_A]->(c)
  WHERE t.fecha >= datetime('2026-08-01') // Busca transferencias recibidas desde el 1 de agosto de 2026.
WITH c, count(t) AS recibidas_ultimo_mes, coalesce(sum(t.monto), 0) AS monto_ultimo_mes // Agrega la cantidad y volumen monetario reciente.
WITH c, recibidas_ultimo_mes, monto_ultimo_mes,
     // mayor cantidad de clientes distintos en alguno de sus dispositivos
     reduce(m = 0, x IN [(c)-[:USA_DISPOSITIVO]->(d:Dispositivo) | // Revisa cada dispositivo o IP al que se conecta la cuenta y cuenta cuántos clientes distintos lo // utilizan (COUNT { MATCH ... }). Luego, el reduce encuentra el valor máximo (max_clientes_dispositivo y max_clientes_ip).
         COUNT { MATCH (d)<-[:USA_DISPOSITIVO]-(:Cuenta)<-[:POSEE]-(cl:Cliente) RETURN DISTINCT cl }]
       | CASE WHEN x > m THEN x ELSE m END) AS max_clientes_dispositivo,
     // mayor cantidad de clientes distintos en alguna de sus IPs
     reduce(m = 0, x IN [(c)-[:CONECTA_DESDE]->(ip:IP) |
         COUNT { MATCH (ip)<-[:CONECTA_DESDE]-(:Cuenta)<-[:POSEE]-(cl:Cliente) RETURN DISTINCT cl }]
       | CASE WHEN x > m THEN x ELSE m END) AS max_clientes_ip, 
     // ¿el dinero sale y vuelve a la cuenta en 3-4 saltos, en orden de fechas?
     EXISTS {
       MATCH p = (c)-[:TRANSFIERE_A*3..4]->(c) //Verifica mediante un booleano si la cuenta forma parte de un ciclo de lavado de dinero de 3 o 4 saltos donde cada 
// transferencia se realizó en una fecha posterior a la anterior (fecha < fecha + 1).
       WHERE all(i IN range(0, length(p) - 2)
                 WHERE relationships(p)[i].fecha < relationships(p)[i + 1].fecha)
     } AS en_ciclo
WITH c, recibidas_ultimo_mes, monto_ultimo_mes, max_clientes_dispositivo, max_clientes_ip, en_ciclo,
     CASE WHEN en_ciclo THEN 30 ELSE 0 END AS p_ciclo,
     CASE WHEN max_clientes_dispositivo >= 9 THEN 30
          WHEN max_clientes_dispositivo >= 7 THEN 15 ELSE 0 END AS p_dispositivo,
     CASE WHEN max_clientes_ip >= 16 THEN 20
          WHEN max_clientes_ip >= 12 THEN 10 ELSE 0 END AS p_ip,
     CASE WHEN monto_ultimo_mes >= 1000000 THEN 20
          WHEN monto_ultimo_mes >= 500000  THEN 10 ELSE 0 END AS p_monto
RETURN c.cuenta_id AS cuenta,
       p_ciclo + p_dispositivo + p_ip + p_monto AS riesgo,
       CASE WHEN p_ciclo + p_dispositivo + p_ip + p_monto >= 60 THEN 'ALTO'
            WHEN p_ciclo + p_dispositivo + p_ip + p_monto >= 30 THEN 'MEDIO'
            ELSE 'BAJO' END AS nivel,
       en_ciclo, max_clientes_dispositivo, max_clientes_ip,
       recibidas_ultimo_mes, round(monto_ultimo_mes) AS monto_ultimo_mes
ORDER BY riesgo DESC, monto_ultimo_mes DESC
LIMIT 20; // Puntuación por ciclo: 30 pts si está en un ciclo temporal. Dispositivo compartido: 30 pts (>= 9 clientes), 15 pts (>= 7 clientes). IP compartida: 20 pts // (>= 16 clientes), 10 pts (>= 12$ clientes). Monto mensual: 20 pts (>= 1000000 clientes), 10 pts (>= 500000 clientes). 
// Clasifica en ALTO (>= 60), MEDIO (>= 30) o BAJO en cualquier otro caso. y luego muestra las 20 cuentas más críticas del sistema. 

// 7. GRAFO de las 5 cuentas más riesgosas (vista Graph)
// Ejecuta exactamente la misma lógica de cálculo de puntaje que el ranking tabular, ordenando y limitando el resultado a las 5 cuentas con mayor riesgo

MATCH (c:Cuenta)
OPTIONAL MATCH (:Cuenta)-[t:TRANSFIERE_A]->(c)
  WHERE t.fecha >= datetime('2026-08-01')
WITH c, coalesce(sum(t.monto), 0) AS monto_ultimo_mes
WITH c, monto_ultimo_mes,
     reduce(m = 0, x IN [(c)-[:USA_DISPOSITIVO]->(d:Dispositivo) |
         COUNT { MATCH (d)<-[:USA_DISPOSITIVO]-(:Cuenta)<-[:POSEE]-(cl:Cliente) RETURN DISTINCT cl }]
       | CASE WHEN x > m THEN x ELSE m END) AS max_disp,
     reduce(m = 0, x IN [(c)-[:CONECTA_DESDE]->(ip:IP) |
         COUNT { MATCH (ip)<-[:CONECTA_DESDE]-(:Cuenta)<-[:POSEE]-(cl:Cliente) RETURN DISTINCT cl }]
       | CASE WHEN x > m THEN x ELSE m END) AS max_ip,
     EXISTS {
       MATCH p = (c)-[:TRANSFIERE_A*3..4]->(c)
       WHERE all(i IN range(0, length(p) - 2)
                 WHERE relationships(p)[i].fecha < relationships(p)[i + 1].fecha)
     } AS en_ciclo
WITH c, monto_ultimo_mes,
     CASE WHEN en_ciclo THEN 30 ELSE 0 END
   + CASE WHEN max_disp >= 9 THEN 30 WHEN max_disp >= 7 THEN 15 ELSE 0 END
   + CASE WHEN max_ip >= 16 THEN 20 WHEN max_ip >= 12 THEN 10 ELSE 0 END
   + CASE WHEN monto_ultimo_mes >= 1000000 THEN 20
          WHEN monto_ultimo_mes >= 500000 THEN 10 ELSE 0 END AS riesgo
ORDER BY riesgo DESC, monto_ultimo_mes DESC
LIMIT 5
// ciclos con fechas en orden (estructura + propiedades)
OPTIONAL MATCH ciclo = (c)-[:TRANSFIERE_A*3..4]->(c) // Extrae las rutas de los ciclos cronológicos donde participa la cuenta.
  WHERE all(i IN range(0, length(ciclo) - 2)
            WHERE relationships(ciclo)[i].fecha < relationships(ciclo)[i + 1].fecha)
WITH c, collect(ciclo) AS ciclos
// transferencias recibidas desde agosto
OPTIONAL MATCH recibe = (:Cuenta)-[t2:TRANSFIERE_A]->(c) // Extrae los enlaces de las transferencias entrantes desde agosto.
  WHERE t2.fecha >= datetime('2026-08-01')
WITH c, ciclos, collect(recibe) AS recibidas
// dispositivos e IPs que usa
OPTIONAL MATCH uso = (c)-[:USA_DISPOSITIVO|CONECTA_DESDE]->() // Trae los dispositivos e IPs a los que la cuenta está conectada.
WITH c, ciclos, recibidas, collect(uso) AS usos
// dueño de la cuenta
OPTIONAL MATCH dueno = (:Cliente)-[:POSEE]->(c) // Agrega el cliente titular de la cuenta.
RETURN c, ciclos, recibidas, usos, collect(dueno) AS duenos;

// V2. Ciclos sospechosos de transferencias (fechas en orden, montos que bajan).
MATCH p = (a:Cuenta)-[:TRANSFIERE_A*3..4]->(a) // Detecta ciclos de dinero que salen y vuelven a la cuenta inicial en 3 o 4 pasos.
WITH p, relationships(p) AS rs, nodes(p)[1..-1] AS internos, a //Aísla los nodos intermediarios del camino.
WHERE all(n IN internos WHERE single(m IN internos WHERE m = n))
  AND NOT a IN internos // Asegura que el ciclo sea simple (sin subbucles repetidos).
  AND all(n IN internos WHERE a.cuenta_id < n.cuenta_id) // Rompe simetrías para evitar devolver el mismo ciclo duplicado con diferente punto de inicio.
  AND all(i IN range(0, size(rs) - 2) WHERE
        rs[i].fecha < rs[i + 1].fecha // Garantiza causalidad en el tiempo (el dinero no puede reenviarse antes de recibirse).
        AND rs[i + 1].monto < rs[i].monto //Valida que el monto se reduzca ligeramente entre cada salto (hasta un 10% por comisión de intermediarios), un indicio
// típico de lavado de dinero estructurado.
        AND rs[i + 1].monto >= rs[i].monto * 0.9)
RETURN p;
